int poserQuestion(int bonneReponse) {
	int reponse;
	int essais = 0;
	do {
		printf("Entrez votre rC)ponse : ");
		scanf("%d", &reponse);
		essais++;
		if (reponse == bonneReponse) {
			printf("Bravo !\n");
			if (essais == 1) return 10;
			else if (essais == 2) return 5;
			else if (essais == 3) return 1;
		} else {
			printf("Mauvaise rC)ponse.\n");
		}
	} while (essais < 3);
	printf("La bonne rC)ponse C)tait : %d\n", bonneReponse);
	return 0;
}

	int a = rand() % 101;

	printf("%d + %d = ?\n", a, b);

	int a = rand() % 11;

	printf("%d * %d = ?\n", a, b);
	*score += poserQuestion(a * b);
}
void jouerSoustraction(int *score) {
	int a = rand() % 101;
	int b = rand() % 101;
	if (a < b) {
		int temp = a;
		a = b;
		b = temp;
	}
	printf("Soustraction\n");
	printf("%d - %d = ?\n", a, b);
	*score += poserQuestion(a - b);
}
void jouerTables(int *score) {
	int tables;
	printf("Quelle table voulez-vous ? (1 C  10) : ");
	scanf("%d", &tables);
	if (tables < 1 || tables > 10) {
		printf("Rentrez une table entre 1 et 10.\n");
		return;
	}
	afficherTables(tables);
	exercerTables(tables, score);
}
void afficherTables(int tables) {
	printf("La table de %d est :\n", tables);
	for (int i = 0; i <= 10; i++) {
		printf("%d * %d = %d\n", tables, i, tables * i);
	}
}
void exercerTables(int tables, int *score) {
	printf("A vous de donner la table de %d\n", tables);
	for (int i = 0; i <= 10; i++) {
		int bonneReponse = tables * i;
		printf("%d * %d = ", tables, i);
		*score += poserQuestion(bonneReponse);
	}
}
void jouerDivisions(int *score) {
	int b = rand() % 10 + 1;
	int resultat = rand() % 11;
	int a = b * resultat;
	printf("Division\n");
	printf("%d / %d = ?\n", a, b);
	*score += poserQuestion(resultat);
}
void chargerScore(char *nom, int *score) {
	FILE *fichier = fopen("scores.txt", "r");
	char nomFichier[TAILLE_NOM];
	int scoreLu;
	if (fichier == NULL) {
		printf("Pas de fichier de scores existant.\n");
		return;
	}
	while (fscanf(fichier, "%s %d", nomFichier, &scoreLu) != EOF) {
		if (strcmp(nomFichier, nom) == 0) {
			*score = scoreLu;
			break;
		}
	}
	fclose(fichier);
}
void sauvegarderScore(char *nom, int score) {
	FILE *fichier = fopen("scores.txt", "r");
	FILE *temp = fopen("temp.txt", "w");
	char nomFichier[TAILLE_NOM];
	int scoreLu;
	int trouve = 0;
	if (fichier == NULL || temp == NULL) {
		printf("Erreur d'accC(s aux fichiers.\n");
		return;
	}
	while (fscanf(fichier, "%s %d", nomFichier, &scoreLu) != EOF) {
		if (strcmp(nomFichier, nom) == 0) {
			fprintf(temp, "%s %d\n", nom, score);
			trouve = 1;
		} else {
			fprintf(temp, "%s %d\n", nomFichier, scoreLu);
		}
	}
	if (!trouve) {
		fprintf(temp, "%s %d\n", nom, score);
	}
	fclose(fichier);
	fclose(temp);
	remove("scores.txt");
	rename("temp.txt", "scores.txt");
}
